(define t%
  (class object%
    (public t1 t2)
    (define t1
      (lambda x
        (printf "t1 called with args: ~a~n" x)))
    
    (define t2

        (printf "t2 called with args: ~a~n" x)

          (send this t1 . y))))
    (super-instantiate ())))

(define t (instantiate t% ()))
(send t t2 'a 'b 'c 'd)

How can I rewrite t2 without using the embedded let?
