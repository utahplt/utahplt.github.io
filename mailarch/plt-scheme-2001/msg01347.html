The code:

(require (lib "url.ss" "net"))
(require (lib "process.ss"))

(define (copy-port in out)
  (let loop ()
    (let ((c (read-char in)))
      (unless (eof-object? c)
        (write-char c out)
        (loop)))))

(define (copy-port2 in out)
  (let loop ()
    (let wait ((r (char-ready? in)))
      (if (not r)
          (wait (char-ready? in))
          (let ((c (read-char in)))
            (unless (eof-object? c)
              (write-char c out)
              (loop)))))))
