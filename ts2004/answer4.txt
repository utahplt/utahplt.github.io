    (define (count-faucets p)
       (cond

        [(boolean? p)            1                ]
       
        [(straight? p) 
              (count-faucets (straight-next p))
          

                                                    ]

        [(branch? p)
           (+ (count-faucets (branch-left p))
              (count-faucets (branch-right p)))
                                                    ]))

     (define (water-running? p)
       (cond

        [(boolean? p)            .p                 ]
       
        [(straight? p) 
              (water-running? (straight-next p))
                                                    ]

        [(branch? p)
          (or (water-running? (branch-left p))
              (water-running? (branch-right p)))
                                                    ]))
